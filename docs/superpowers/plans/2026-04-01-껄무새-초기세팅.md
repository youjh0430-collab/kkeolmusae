```typescript
/**
 * 수익률: (현재가 - 매수가) / 매수가 × 100
 */
export function calculateReturnRate(buyPrice: number, currentPrice: number): number {
  return ((currentPrice - buyPrice) / buyPrice) * 100
}

/**
 * 투자 가능 금액: 소비 항목 단가 × 기간 일수
 */
export function calculateInvestmentAmount(itemPrice: number, periodDays: number): number {
  return itemPrice * periodDays
}

/**
 * 수익금: 투자 금액 × (수익률 / 100)
 */
export function calculateProfitAmount(investmentAmount: number, returnRate: number): number {
  return investmentAmount * (returnRate / 100)
}

/**
 * 두 날짜 사이 일수 (당일 미포함)
 */
export function getPeriodDays(periodStart: string, periodEnd: string): number {
  const start = new Date(periodStart)
  const end = new Date(periodEnd)
  return Math.round((end.getTime() - start.getTime()) / (1000 * 60 * 60 * 24))
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import yahooFinance from 'yahoo-finance2'
import type { StockHistoryResponse } from '@/types'

// GET /api/stock/history?ticker=AAPL&from=2024-01-01&to=2024-12-31
export async function GET(request: NextRequest) {
  const { searchParams } = new URL(request.url)
  const ticker = searchParams.get('ticker')
  const from = searchParams.get('from')
  const to = searchParams.get('to')

  if (!ticker || !from || !to) {
    return NextResponse.json(
      { error: 'ticker, from, to 파라미터가 필요합니다.' },
      { status: 400 }
    )
  }

  try {
    const result = await yahooFinance.chart(ticker, {
      period1: from,
      period2: to,
      interval: '1d',
    })

    const history = (result.quotes ?? [])
      .filter((q) => q.close !== null)
      .map((q) => ({
        date: new Date(q.date).toISOString().split('T')[0],
        close: q.close as number,
      }))

    const response: StockHistoryResponse = { ticker, history }
    return NextResponse.json(response)
  } catch (error) {
    console.error('[stock/history] error:', error)
    return NextResponse.json(
      { error: '주가 데이터를 가져오지 못했습니다.' },
      { status: 502 }
    )
  }
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server'
import yahooFinance from 'yahoo-finance2'
import { supabaseAdmin } from '@/lib/supabase'
import {
  calculateReturnRate,
  calculateInvestmentAmount,
  calculateProfitAmount,
  getPeriodDays,
} from '@/lib/calculate'
import type { SimulationInput } from '@/types'

export async function POST(request: NextRequest) {
  let body: SimulationInput

  try {
    body = await request.json()
  } catch {
    return NextResponse.json({ error: '잘못된 요청 형식입니다.' }, { status: 400 })
  }

  const { itemName, itemPrice, periodStart, periodEnd, stockTicker, stockName, userId } = body

  if (!itemName || !itemPrice || !periodStart || !periodEnd || !stockTicker || !stockName) {
    return NextResponse.json({ error: '필수 필드가 누락되었습니다.' }, { status: 400 })
  }

  try {
    // 1. 시작일 종가 조회
    const startChart = await yahooFinance.chart(stockTicker, {
      period1: periodStart,
      period2: periodStart,
      interval: '1d',
    })
    const startQuotes = (startChart.quotes ?? []).filter((q) => q.close !== null)
    if (startQuotes.length === 0) {
      return NextResponse.json({ error: '시작일 주가 데이터가 없습니다.' }, { status: 422 })
    }
    const buyPrice = startQuotes[0].close as number

    // 2. 현재 주가 조회
    const quote = await yahooFinance.quote(stockTicker)
    const currentPrice = quote.regularMarketPrice ?? quote.regularMarketPreviousClose
    if (!currentPrice) {
      return NextResponse.json({ error: '현재 주가 데이터가 없습니다.' }, { status: 422 })
    }

    // 3. 계산
    const periodDays = getPeriodDays(periodStart, periodEnd)
    const investmentAmount = calculateInvestmentAmount(itemPrice, periodDays)
    const returnRate = calculateReturnRate(buyPrice, currentPrice)
    const profitAmount = calculateProfitAmount(investmentAmount, returnRate)

    // 4. Supabase 저장
    const { data, error } = await supabaseAdmin
      .from('simulations')
      .insert({
        user_id: userId ?? null,
        item_name: itemName,
        item_price: itemPrice,
        period_start: periodStart,
        period_end: periodEnd,
        stock_ticker: stockTicker,
        stock_name: stockName,
        buy_price: buyPrice,
        current_price: currentPrice,
        return_rate: returnRate,
        investment_amount: investmentAmount,
        profit_amount: profitAmount,
      })
      .select()
      .single()

    if (error) {
      console.error('[simulate] supabase error:', error)
      return NextResponse.json({ error: 'DB 저장 실패' }, { status: 500 })
    }

    return NextResponse.json(data, { status: 201 })
  } catch (error) {
    console.error('[simulate] error:', error)
    return NextResponse.json({ error: '시뮬레이션 처리 중 오류가 발생했습니다.' }, { status: 502 })
  }
}
```
